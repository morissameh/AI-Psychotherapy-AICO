import firebase_admin
from firebase_admin import credentials, firestore

# Use a service account
cred = credentials.Certificate('graduation-project-c4f4c-firebase-adminsdk-xi9p6-6dc12a5470.json')
firebase_admin.initialize_app(cred)

db = firestore.client()

def add_patient_info(patient_name, question, answer):
    # Reference to the specific document
    patient_doc = db.collection('patients').document(patient_name)
    
    # Get the existing document data
    doc = patient_doc.get()
    
    if doc.exists:
        # If document exists, update it with new question and answer
        existing_data = doc.to_dict()
        existing_data[question] = answer
        patient_doc.set(existing_data)
        print("Patient info updated!")
    else:
        # If document does not exist, create it with new information
        patient_info = {
            question: answer
        }
        patient_doc.set(patient_info)
        print("Patient info added!")

def get_patient_info(patient_id):
    # Reference to the specific document
    patient_doc = db.collection('patients').document(patient_id)
    
    # Get the document
    doc = patient_doc.get()
    
    if doc.exists:
        print(f"Document data: {doc.to_dict()}")
    else:
        print("No such document!")

# Example usage
patient_name = "moris"
add_patient_info(patient_name, "what is your name?", "my name is moris")
add_patient_info(patient_name, "how old are you?", "I have 23 years old")

# Retrieve and print patient info
get_patient_info(patient_name)
